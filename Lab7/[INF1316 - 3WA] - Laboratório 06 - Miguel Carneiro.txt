CORRIDASAPOS.C
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

#define NUM_THREADS 5
#define PULO_MAXIMO 100
#define DESCANSO_MAXIMO 1
#define DISTANCIA_PARA_CORRER 100

static int classificacao = 1;
static pthread_mutex_t lock;
static char * resp[200]; 
static int cont = 0;

void *Correr(void *sapo);

int main(void)
{

    classificacao =1;
    pthread_t threads[NUM_THREADS];
    int t;
    printf("Corrida iniciada ... \n");
    for(t=0;t < NUM_THREADS;t++)
        pthread_create(&threads[t], NULL, Correr, (void *) t);
    for(t=0;t < NUM_THREADS; t++)
        pthread_join(threads[t],NULL);
    printf("\n Acabou!!\n");
    pthread_exit(NULL); 

    return 0;
}

void *Correr(void *sapo)
{
    int pulos = 0;
    int distanciaJaCorrida = 0;

    while (distanciaJaCorrida <= DISTANCIA_PARA_CORRER) 
    {
        int pulo = rand() % PULO_MAXIMO;
        distanciaJaCorrida += pulo;
        pulos++;
        printf("Sapo %d pulou\n", (int) sapo);
        int descanso = rand() % DESCANSO_MAXIMO;
        sleep(descanso);
    }

    printf("Sapo %d chegou na posicao %d com %d pulos\n", (int) sapo, classificacao, pulos);
    cont++;
    classificacao++;
    pthread_exit(NULL); 
}

QUESTAO 3) Usando threads, escreva um programa C que implemente o problema do produtor/consumidor. O produtor deve produzir dados (numeros inteiros pseudo-aleatorios) a cada 1 segundo colocando-os em uma fila (buffer, implementacao circular). O consumidor deve retirar dados da fila a cada 2 segundos. O tamanho maximo da fila deve ser de 8 elementos (MAXFILA) e tanto o produtor como o consumidor devem produzir/consumir 64 elementos (numeros inteiros de 1 a 64, por exemplo) evitando condicoes de corrida. Variaveis compartilhadas entre threads sao simplesmente variaveis globais.
-----------------------------------------------------
PRODUTORCONSUMIDOR.C
#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

#define MAXFILA 8
#define TOTAL 64

static int buffer[MAXFILA];
static int inicio = 0;
static int fim = 0;
static int qtd = 0;

static pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t naoCheio = PTHREAD_COND_INITIALIZER;
static pthread_cond_t naoVazio = PTHREAD_COND_INITIALIZER;

void *Produtor(void *arg);
void *Consumidor(void *arg);

int main(void)
{
    pthread_t produtor, consumidor;

    printf("Inicio\n");
    pthread_create(&produtor, NULL, Produtor, NULL);
    pthread_create(&consumidor, NULL, Consumidor, NULL);
    pthread_join(produtor, NULL);
    pthread_join(consumidor, NULL);
    printf("\nFim!!\n");

    pthread_mutex_destroy(&mutex);
    pthread_cond_destroy(&naoCheio);
    pthread_cond_destroy(&naoVazio);
    return 0;
}

void *Produtor(void *arg)
{
    for (int i = 1; i <= TOTAL; i++)
    {
        sleep(1);
        int valor = rand() % 100;

        pthread_mutex_lock(&mutex);
        while (qtd == MAXFILA)
            pthread_cond_wait(&naoCheio, &mutex);

        buffer[fim] = valor;
        fim = (fim + 1) % MAXFILA;
        qtd++;
        printf("Produtor:   %2do valor = %2d | fila com %d\n", i, valor, qtd);

        pthread_cond_signal(&naoVazio);
        pthread_mutex_unlock(&mutex);
    }
    pthread_exit(NULL);
}

void *Consumidor(void *arg)
{
    for (int i = 1; i <= TOTAL; i++)
    {
        sleep(2);

        pthread_mutex_lock(&mutex);
        while (qtd == 0)
            pthread_cond_wait(&naoVazio, &mutex);

        int valor = buffer[inicio];
        inicio = (inicio + 1) % MAXFILA;
        qtd--;
        printf("Consumidor: %2do valor = %2d | fila com %d\n", i, valor, qtd);

        pthread_cond_signal(&naoCheio);
        pthread_mutex_unlock(&mutex);
    }
    pthread_exit(NULL);
}

O buffer e um vetor global de MAXFILA = 8 posicoes usado como fila circular. A variavel fim indica onde o produtor insere, inicio indica de onde o consumidor retira, e qtd guarda quantos elementos estao na fila. Os indices andam com (indice + 1) % MAXFILA, entao ao chegar no fim do vetor eles voltam para a posicao 0. O produtor gera um numero aleatorio a cada 1 segundo e o consumidor retira um a cada 2 segundos, cada um 64 vezes.

Como buffer, inicio, fim e qtd sao compartilhados, todo acesso a eles e feito com o mutex travado, evitando condicoes de corrida. Alem disso foram usadas duas variaveis de condicao: naoCheio e naoVazio. Se a fila esta cheia, o produtor chama pthread_cond_wait(&naoCheio, &mutex), que libera o mutex e bloqueia a thread ate o consumidor retirar um elemento e sinalizar naoCheio. Do mesmo jeito, se a fila esta vazia, o consumidor espera em naoVazio ate o produtor inserir algo. O wait fica dentro de um while para que a condicao seja testada de novo quando a thread acordar.

No resultado da para ver o comportamento esperado: no inicio o produtor e mais rapido e a fila vai enchendo ate chegar em 8. A partir dai o produtor fica bloqueado e so consegue inserir quando o consumidor retira um elemento, por isso as linhas passam a alternar com a fila oscilando entre 7 e 8. Depois que o produtor termina os 64 elementos, o consumidor esvazia a fila ate 0. Os valores sao consumidos exatamente na mesma ordem em que foram produzidos, sem perda nem repeticao.

QUESTAO 4) Modifique o programa anterior, para que haja 2 (ou mais) threads consumidor e 2 (ou mais) threads produtor. O que muda em relacao ao uso do mutex e da variavel de condicao?
-----------------------------------------------------
PRODUTORCONSUMIDORMULTI.C
#include <pthread.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

#define MAXFILA 8
#define TOTAL 64
#define NUM_PRODUTORES 2
#define NUM_CONSUMIDORES 2

static int buffer[MAXFILA];
static int inicio = 0;
static int fim = 0;
static int qtd = 0;

static pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;
static pthread_cond_t naoCheio = PTHREAD_COND_INITIALIZER;
static pthread_cond_t naoVazio = PTHREAD_COND_INITIALIZER;

void *Produtor(void *arg);
void *Consumidor(void *arg);

int main(void)
{
    pthread_t produtores[NUM_PRODUTORES];
    pthread_t consumidores[NUM_CONSUMIDORES];
    int t;

    printf("Inicio\n");
    for (t = 0; t < NUM_PRODUTORES; t++)
        pthread_create(&produtores[t], NULL, Produtor, (void *)(intptr_t) t);
    for (t = 0; t < NUM_CONSUMIDORES; t++)
        pthread_create(&consumidores[t], NULL, Consumidor, (void *)(intptr_t) t);

    for (t = 0; t < NUM_PRODUTORES; t++)
        pthread_join(produtores[t], NULL);
    for (t = 0; t < NUM_CONSUMIDORES; t++)
        pthread_join(consumidores[t], NULL);
    printf("\nFim!!\n");

    pthread_mutex_destroy(&mutex);
    pthread_cond_destroy(&naoCheio);
    pthread_cond_destroy(&naoVazio);
    return 0;
}

void *Produtor(void *arg)
{
    int id = (int)(intptr_t) arg;

    for (int i = 1; i <= TOTAL / NUM_PRODUTORES; i++)
    {
        sleep(1);
        int valor = rand() % 100;

        pthread_mutex_lock(&mutex);

        while (qtd == MAXFILA)
            pthread_cond_wait(&naoCheio, &mutex);

        buffer[fim] = valor;
        fim = (fim + 1) % MAXFILA;
        qtd++;
        printf("Produtor %d:   valor = %2d | fila com %d\n", id, valor, qtd);

        pthread_cond_signal(&naoVazio);
        pthread_mutex_unlock(&mutex);
    }
    pthread_exit(NULL);
}

void *Consumidor(void *arg)
{
    int id = (int)(intptr_t) arg;

    for (int i = 1; i <= TOTAL / NUM_CONSUMIDORES; i++)
    {
        sleep(2);

        pthread_mutex_lock(&mutex);
        while (qtd == 0)
            pthread_cond_wait(&naoVazio, &mutex);

        int valor = buffer[inicio];
        inicio = (inicio + 1) % MAXFILA;
        qtd--;
        printf("Consumidor %d: valor = %2d | fila com %d\n", id, valor, qtd);

        pthread_cond_signal(&naoCheio);
        pthread_mutex_unlock(&mutex);
    }
    pthread_exit(NULL);
}

CONCLUSAO
Foram criadas 2 threads produtoras e 2 consumidoras, cada uma recebendo um id como argumento. Para manter o total de 64 elementos, cada produtor produz TOTAL / NUM_PRODUTORES = 32 valores e cada consumidor consome TOTAL / NUM_CONSUMIDORES = 32 valores. O buffer circular, o mutex e as duas variaveis de condicao continuam os mesmos da questao anterior.

Em relacao ao mutex, a diferenca e que agora ele tambem protege threads do mesmo tipo umas das outras. Com um produtor e um consumidor, cada indice (fim e inicio) era alterado por uma unica thread. Agora dois produtores podem tentar escrever em buffer[fim] e avancar fim ao mesmo tempo, e dois consumidores podem ler o mesmo buffer[inicio]. Sem o mutex, dois produtores poderiam gravar na mesma posicao, perdendo um valor, ou dois consumidores poderiam retirar o mesmo elemento.

Em relacao a variavel de condicao, passa a ser obrigatorio testar a condicao em um while e nao em um if. Quando um produtor sinaliza naoVazio, um consumidor acorda, mas antes de ele conseguir pegar o mutex de volta o outro consumidor pode ter retirado o elemento, e a fila volta a ficar vazia. Com o while, a thread que acorda testa de novo e volta a esperar se for o caso. O mesmo vale para os produtores com a fila cheia. Tambem pode ser usado pthread_cond_broadcast no lugar de pthread_cond_signal para acordar todas as threads que estao esperando, mas como cada insercao libera so um elemento e cada retirada libera so uma vaga, o signal junto com o while ja e suficiente. No resultado os valores continuam saindo na mesma ordem em que entraram na fila, a fila nunca passa de 8 nem fica negativa e os 64 elementos sao consumidos sem repeticao.
