QUESTAO 1) Faca um programa que usa myallocate para alocar e desalocar memoria dinamicamente.

MYALLOCATE.C
#include<stdio.h>
#include<unistd.h>

void *memorypool;

void *myallocate(int n);
void initmemorypool(void);
void resetmemorypool(void);

int main(void)
{
    printf("---  Limite inicial do heap %p -----\n", sbrk(0));
    initmemorypool();

    int* v1 = (int*) myallocate(sizeof(int)*4);
    if (v1 == NULL)
    {
        printf("Erro na alocacao de momeria pro v1");
        return 1;
    }

    printf("--- Novo limite para o heap %p -----\n", sbrk(0));
    resetmemorypool();
    return 0;
}

void *myallocate( int n){
    void* ptr = sbrk(n);
    if(ptr == (void*)-1)
    {
        return NULL;
    }
    return ptr;
}
void initmemorypool(void)
{
    memorypool = sbrk(0);
}
void resetmemorypool(void)
{
    brk(memorypool);
}

CONCLUSAO
O programa usa as funcoes do myallocate.c para gerenciar o heap diretamente com sbrk() e brk(), sem passar pelo malloc(). A funcao initmemorypool() guarda em memorypool o limite atual do heap (sbrk(0)). Depois, myallocate(n) chama sbrk(n), que empurra o limite do heap n bytes para frente e retorna o endereco antigo, que passa a ser o inicio da area alocada (aqui 16 bytes, um vetor de 4 inteiros). Se o sbrk falhar ele retorna (void*)-1, e nesse caso myallocate retorna NULL. Por fim, resetmemorypool() chama brk(memorypool), que move o limite do heap de volta para onde estava, desalocando de uma vez tudo o que foi alocado depois do init. Por isso essa forma de alocacao funciona como uma pilha: nao da para liberar uma alocacao especifica, so voltar o limite para um ponto anterior.

Um detalhe interessante aparece na saida: a diferenca entre os dois enderecos impressos e de 0x22010 bytes (cerca de 136 KB), e nao apenas os 16 bytes que foram alocados. Isso acontece porque o sbrk(0) do primeiro printf e avaliado antes do printf executar, e o proprio printf, na primeira vez que e chamado, usa malloc() para criar o buffer da saida padrao. O malloc chama brk() internamente e aumenta o heap em um bloco grande. So depois disso o initmemorypool() pega o limite do heap, e entao o myallocate soma os 16 bytes (o final ...010 do endereco). Isso mostra na pratica o aviso do laboratorio de que alocar com sbrk/brk pode conflitar com chamadas que usam malloc(), ja que os dois mexem no mesmo limite do heap.

MAP-EXEMPLO.C
#include <stdio.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>
#include <stdlib.h>

int main (int argc, char *argv[])
{
    struct stat sb;
    off_t len;
    char *p;
    int fd;

    if (argc < 2) {
        fprintf (stderr, "usage: %s <file>\n", argv[0]);
        exit(1);
    }

    fd = open (argv[1], O_RDONLY);
    if (fd == -1) {
        perror ("open");
        exit(1);
    }
    if (fstat(fd, &sb)== -1) {
        perror ("fstat");
        exit(1);
    }

    if (!S_ISREG (sb.st_mode)) {
        fprintf (stderr, "%s is not a file\n", argv[1]);
        exit(1);
    }
    p = mmap (0, sb.st_size, PROT_READ, MAP_SHARED, fd, 0);
    if (p == MAP_FAILED) {
        perror ("mmap"); exit(1);
    }
    if (close (fd) == -1) {
        perror ("close"); exit(1);
    }
    for (len = 0; len < sb.st_size; len++)
        putchar (p[len]);
    if (munmap (p, sb.st_size) == -1) {
        perror ("munmap");
        exit(1);
    }
    return 0;
}

QUESTAO 3) Modifique map-exemplo.c para: a. Verificar quantas paginas de memoria o seu arquivo precisou. b. Medir o quao mais rapido foi a escrita do arquivo mapeado quando comparado com a E/S em disco. Dica: use gettimeofday()

NOVOMAP-EXEMPLO.C
#include <stdio.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>
#include <stdlib.h>
#include<sys/time.h>

int main (int argc, char *argv[])
{
    struct stat sb;
    off_t len;
    char *p;
    int fd;
    struct timeval inicio, fim;
    long tempo_mmap, tempo_disco;

    if (argc < 2) {
        fprintf (stderr, "usage: %s <file>\n", argv[0]);
        exit(1);
    }

    fd = open (argv[1], O_RDONLY);
    if (fd == -1) {
        perror ("open");
        exit(1);
    }
    if (fstat(fd, &sb)== -1) {
        perror ("fstat");
        exit(1);
    }

    if (!S_ISREG (sb.st_mode)) {
        fprintf (stderr, "%s is not a file\n", argv[1]);
        exit(1);
    }
    p = mmap (0, sb.st_size, PROT_READ, MAP_SHARED, fd, 0);
    if (p == MAP_FAILED) {
        perror ("mmap"); exit(1);
    }

    gettimeofday(&inicio, NULL);
    for (len = 0; len < sb.st_size; len++)
        putchar (p[len]);
    gettimeofday(&fim, NULL);
    tempo_mmap = (fim.tv_sec - inicio.tv_sec) * 1000000L + (fim.tv_usec - inicio.tv_usec);

    long pagesize = sysconf(_SC_PAGESIZE);
    long pages = (sb.st_size + pagesize -1) / pagesize;

    lseek(fd, 0, SEEK_SET); // garante que o cursor está no início do arquivo
    char c;
    gettimeofday(&inicio, NULL);
    for (len = 0; len < sb.st_size; len++) {
        read(fd, &c, 1);
        putchar(c);
    }
    gettimeofday(&fim, NULL);
    if (close (fd) == -1) {
        perror ("close"); exit(1);
    }
    tempo_disco = (fim.tv_sec - inicio.tv_sec) * 1000000L + (fim.tv_usec - inicio.tv_usec);
    printf("\nTempo com mmap:  %ld us\n", tempo_mmap);
    printf("Tempo com disco: %ld us\n", tempo_disco);

    printf("Tamanho do arquivo: %ld bytes\n", (long)sb.st_size);
    printf("Tamanho da pagina:  %ld bytes\n", pagesize);
    printf("Paginas utilizadas: %ld\n", pages);

    if (munmap (p, sb.st_size) == -1) {
        perror ("munmap");
        exit(1);
    }
    return 0;
}

QUESTAO 4) Escreva dois programas prog1.c e prog2.c que trocam dados atraves de um mesmo arquivo mapeado para memoria (de cada um).
-----------------------------------------------------
PROG1.C
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>

#define TAM 100

int main(int argc, char *argv[])
{

    if (argc < 2) {
        fprintf (stderr, "usage: %s <file>\n", argv[0]);
        exit(1);
    }

    int fd = open(argv[1], O_RDWR | O_CREAT, 0666);
    if (fd == -1) {
        perror ("open");
        exit(1);
    }
    if (ftruncate(fd, TAM) == -1) {
        perror ("ftruncate");
        exit(1);
    }
    char *p = (char*) mmap(NULL, TAM, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (p == MAP_FAILED) {
        perror ("mmap");
        exit(1);
    }
    close(fd);

    strcpy(p, "Mensagem enviada pelo prog1!\n");
    printf("[prog1] Escrevi no arquivo mapeado: \"%s\"", p);
    printf("[prog1] Aguardando prog2 responder...\n");

    while (strcmp(p, "Mensagem enviada pelo prog1!\n") == 0) {
        sleep(1);
    }
    printf("[prog1] Recebi resposta do prog2: \"%s\"\n", p);

    munmap(p, TAM);
    return 0;
}

CONCLUSAO
Os dois programas se comunicam mapeando o mesmo arquivo na memoria com a flag MAP_SHARED. Com essa flag, as paginas mapeadas sao compartilhadas entre todos os processos que mapeiam o arquivo, entao o que um processo escreve na sua regiao aparece imediatamente na regiao do outro, sem nenhum read() ou write(). Se fosse usado MAP_PRIVATE, cada processo teria a sua propria copia e a troca de mensagens nao funcionaria.

O prog1 abre (ou cria) o arquivo e usa ftruncate() para garantir que ele tenha exatamente 100 bytes (TAM). Isso e necessario porque um arquivo recem-criado tem tamanho 0 e nao e possivel mapear nem acessar uma regiao alem do fim do arquivo. Depois de mapear, ele escreve a sua mensagem com strcpy() direto na memoria e fica esperando, verificando a cada 1 segundo se o conteudo ainda e a mensagem que ele escreveu. O prog2 mapeia o mesmo arquivo, le a mensagem do prog1 e escreve a resposta por cima. Na proxima verificacao o prog1 percebe que o conteudo mudou e imprime a resposta. Por isso o prog2 tem que ser executado depois do prog1, ja que e o prog1 que cria o arquivo com o tamanho certo. Como o mapeamento e de um arquivo real, a ultima mensagem continua salva no comunicacao.txt depois que os dois programas terminam. Vale notar que a sincronizacao aqui e feita por espera ativa (o laco com sleep), sem nenhum mecanismo de exclusao mutua, entao ela funciona para essa troca simples, mas nao garantiria a consistencia se os dois escrevessem ao mesmo tempo.

QUESTAO 5) Escreva um programa que faz um mapeamento de arquivo para memoria e depois escreve novos caracteres no final do arquivo. O que voce precisa fazer para evitar um SegFault?
-----------------------------------------------------
EX5.C
#include <stdio.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>
#include <stdlib.h>
#include <string.h>

// Bom exercício para entender o funcionamento do mapeamento de arquivo na memória e uso do memcpy com mmap!!!

int main (int argc, char *argv[])
{
    struct stat sb;
    char *p;
    int fd;
    off_t tam_antigo, tam_novo;
    char *texto = "Texto adicionado no final pelo ex5!\n";

    if (argc < 2) {
        fprintf (stderr, "usage: %s <file> [texto]\n", argv[0]);
        exit(1);
    }

    if(argc >= 3)
    {
        texto = argv[2];
    }

    fd = open (argv[1], O_RDWR | O_CREAT, 0666);
    if (fd == -1) {
        perror ("open");
        exit(1);
    }
    if (fstat(fd, &sb) == -1) {
        perror ("fstat");
        exit(1);
    }
    if (!S_ISREG (sb.st_mode)) {
        fprintf (stderr, "%s is not a file\n", argv[1]);
        exit(1);
    }

    tam_antigo = sb.st_size;
    tam_novo = tam_antigo + strlen(texto);

    if (ftruncate(fd, tam_novo) == -1) {
        perror ("ftruncate");
        exit(1);
    }

    p = mmap (0, tam_novo, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (p == MAP_FAILED) {
        perror ("mmap"); exit(1);
    }
    if (close (fd) == -1) {
        perror ("close"); exit(1);
    }

    memcpy(p + tam_antigo, texto, strlen(texto));

    printf("Tamanho antigo: %ld bytes\n", (long)tam_antigo);
    printf("Tamanho novo:   %ld bytes\n", (long)tam_novo);

    if (munmap (p, tam_novo) == -1) {
        perror ("munmap");
        exit(1);
    }
    return 0;
}

Para evitar o erro e preciso aumentar o tamanho do arquivo no disco antes de escrever alem do seu fim, usando ftruncate(), e fazer o mmap() ja com o tamanho novo. O mapeamento so da acesso aos bytes que o arquivo realmente tem, ele nao faz o arquivo crescer sozinho. Se o programa mapeasse uma regiao maior que o arquivo e escrevesse depois do fim, existem dois casos: se a escrita cair no resto da ultima pagina (a parte preenchida com zeros) ela nao gera erro, mas tambem nao e salva no arquivo; e se cair em uma pagina que esta inteira depois do fim do arquivo, o processo e encerrado pelo kernel com um sinal de erro de memoria (SIGBUS no Linux, ou SIGSEGV se o acesso estiver fora da regiao mapeada).

O programa abre o arquivo para leitura e escrita, usa fstat() para descobrir o tamanho atual (tam_antigo) e calcula o tamanho novo somando o tamanho do texto. Em seguida chama ftruncate(fd, tam_novo), que aumenta o arquivo preenchendo os bytes novos com zeros, e so entao mapeia o arquivo inteiro com PROT_READ | PROT_WRITE e MAP_SHARED, para que as escritas sejam refletidas no arquivo. Como o ponteiro p retornado pelo mmap aponta para o inicio do arquivo (offset 0), o texto e copiado com memcpy para p + tam_antigo, ou seja, logo depois do ultimo byte do conteudo original, substituindo os zeros criados pelo ftruncate. O texto pode ser passado como segundo argumento; se nao for passado, o programa usa um texto padrao. Na execucao, o arquivo comecou com 26 bytes, passou para 52 depois da primeira chamada e para 88 depois da segunda, sem perder o conteudo anterior.

QUESTAO 6) Re-escreva o seu programa do item anterior usando mremap()

EX6.C
#define _GNU_SOURCE
#include <stdio.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/mman.h>
#include <stdlib.h>
#include <string.h>

int main (int argc, char *argv[])
{
    struct stat sb;
    char *p;
    int fd;
    off_t tam_antigo, tam_novo;
    char *texto = "Texto adicionado no final pelo ex5!\n";

    if (argc < 2) {
        fprintf (stderr, "usage: %s <file> [texto]\n", argv[0]);
        exit(1);
    }

    if(argc >= 3)
    {
        texto = argv[2];
    }

    fd = open (argv[1], O_RDWR | O_CREAT, 0666);
    if (fd == -1) {
        perror ("open");
        exit(1);
    }
    if (fstat(fd, &sb) == -1) {
        perror ("fstat");
        exit(1);
    }
    if (!S_ISREG (sb.st_mode)) {
        fprintf (stderr, "%s is not a file\n", argv[1]);
        exit(1);
    }

    tam_antigo = sb.st_size;
    if(tam_antigo == 0)
    {
        fprintf(stderr, "Arquivo vazio, use um arquivo existente como base.\n");
        exit(1);
    }
    p = mmap (0, tam_antigo, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
    if (p == MAP_FAILED) {
        perror ("mmap"); exit(1);
    }

    tam_novo = tam_antigo + strlen(texto);

    if (ftruncate(fd, tam_novo) == -1) {
        perror ("ftruncate");
        exit(1);
    }

    p = mremap(p, tam_antigo, tam_novo, MREMAP_MAYMOVE);
    if (p == MAP_FAILED)
    {
        perror("mremap");
        exit(1);
    }
    if (close (fd) == -1) {
        perror ("close"); exit(1);
    }

    memcpy(p + tam_antigo, texto, strlen(texto));

    printf("Tamanho antigo: %ld bytes\n", (long)tam_antigo);
    printf("Tamanho novo:   %ld bytes\n", (long)tam_novo);

    if (munmap (p, tam_novo) == -1) {
        perror ("munmap");
        exit(1);
    }
    return 0;
}

CONCLUSAO
A diferenca para a questao anterior e que agora o arquivo e mapeado primeiro com o seu tamanho original, e so depois o mapeamento e aumentado com mremap(). No ex5 o tamanho final ja era conhecido antes do mmap, entao bastava mapear direto com ele. O mremap() e util justamente quando o mapeamento ja existe e precisa crescer depois, sem ter que fazer munmap() e mmap() de novo. Ele recebe o endereco atual do mapeamento, o tamanho antigo, o tamanho novo e as flags. Foi usada a flag MREMAP_MAYMOVE, que permite ao kernel mover o mapeamento para outro endereco caso nao haja espaco livre logo depois da regiao atual. Por isso o retorno do mremap e atribuido de novo a p, ja que o endereco antigo pode deixar de ser valido. Como mremap() e uma funcao especifica do Linux, foi necessario colocar #define _GNU_SOURCE antes dos includes, senao a funcao e a constante MREMAP_MAYMOVE nao sao declaradas e o programa nao compila.

O mremap() so altera o tamanho da regiao na memoria, ele nao aumenta o arquivo. Por isso continua sendo necessario chamar ftruncate() antes, pelo mesmo motivo da questao 5: sem isso, a parte nova do mapeamento nao teria bytes do arquivo por tras e a escrita causaria erro. Depois disso o texto e copiado para p + tam_antigo com memcpy, e o munmap() e feito com o tamanho novo. Tambem foi tratado o caso de arquivo vazio, porque o mmap() inicial com tamanho 0 falha com EINVAL; nesse caso o programa avisa e encerra, como mostra a ultima execucao com vazio.txt.
